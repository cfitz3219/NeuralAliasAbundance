cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(NeuralAliasCpuRegressions LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(NA_CPU_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
add_library(na_cpu_test_options INTERFACE)
target_compile_features(na_cpu_test_options INTERFACE cxx_std_17)
# Test the recovered headers without imposing new warnings policy on frozen code.
target_include_directories(na_cpu_test_options SYSTEM INTERFACE "${NA_CPU_ROOT}/src")
target_link_libraries(na_cpu_test_options INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(na_cpu_test_options INTERFACE /W4 /WX /permissive- /utf-8)
  target_compile_definitions(na_cpu_test_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(na_cpu_test_options INTERFACE -Wall -Wextra -Werror -pedantic)
endif()
function(na_add_cpu_test name)
  add_executable(na_cpu_${name} ${name}_test.cpp ${ARGN})
  target_link_libraries(na_cpu_${name} PRIVATE na_cpu_test_options)
  add_test(NAME na.cpu.${name} COMMAND na_cpu_${name})
  set_tests_properties(na.cpu.${name} PROPERTIES LABELS "cpu;reconstructed" TIMEOUT 30)
endfunction()
na_add_cpu_test(frame_contract "${NA_CPU_ROOT}/src/core/frame_contract.cpp")

else()

endif()
