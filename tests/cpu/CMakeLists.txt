  target_compile_definitions(na_cpu_test_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(na_cpu_test_options INTERFACE -Wall -Wextra -Werror -pedantic)
endif()
function(na_add_cpu_test name)
  add_executable(na_cpu_${name} ${name}_test.cpp ${ARGN})
  target_link_libraries(na_cpu_${name} PRIVATE na_cpu_test_options)
  add_test(NAME na.cpu.${name} COMMAND na_cpu_${name})
  set_tests_properties(na.cpu.${name} PROPERTIES LABELS "cpu;reconstructed" TIMEOUT 30)
endfunction()
